// Scripts for AI Scheduler Dashboard development and generation

// Build scripts

// Generate chart components using Plop
// Usage: npx plop chart
// Prompts: Chart name (PascalCase), Chart shape

// Generate panel components using Plop  
// Usage: npx plop panel
// Prompts: Panel name (PascalCase), Panel title

// Generate dashboard pages using Plop
// Usage: npx plop dashboard  
// Prompts: Route name (kebab-case), Dashboard title

// Generate primitive components using Plop
// Usage: npx plop primitive
// Prompts: Primitive name (PascalCase), Category

// Development scripts

// Build TypeScript for production
// Usage: npm run build

// Watch for TypeScript changes
// Usage: npm run dev

// Run Vitest test suite
// Usage: npm test

// Run Vitest with UI
// Usage: npm run test:ui

// Main generation script
// Usage: node scripts/generate.js <chart|panel|dashboard>
